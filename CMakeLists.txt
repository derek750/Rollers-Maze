cmake_minimum_required(VERSION 3.20)
project(RollersMaze LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenGL REQUIRED)
find_package(glfw3 3.3 REQUIRED)
find_package(GLEW 2.2 REQUIRED)
add_executable(rollers src/main.cpp src/gl.cpp)
target_link_libraries(rollers PRIVATE glfw GLEW::GLEW OpenGL::GL)
option(ROLLERS_APPLE_SIMD "Use Apple SDK SIMD math for an offline Mac build" OFF)
add_library(board_math INTERFACE)
if(APPLE AND ROLLERS_APPLE_SIMD)
    target_compile_definitions(board_math INTERFACE ROLLERS_APPLE_SIMD)
else()
    find_package(glm CONFIG REQUIRED)
    target_link_libraries(board_math INTERFACE glm::glm)
endif()
target_sources(rollers PRIVATE src/renderer.cpp src/maze.cpp)
target_link_libraries(rollers PRIVATE board_math)
add_library(simulation src/physics.cpp)
target_include_directories(simulation PUBLIC src)
target_link_libraries(simulation PUBLIC board_math)
target_link_libraries(rollers PRIVATE simulation)
include(CTest)
if(BUILD_TESTING)
    add_executable(maze_checks tests/checks.cpp)
    target_link_libraries(maze_checks PRIVATE simulation)
    add_test(NAME maze_checks COMMAND maze_checks)
endif()
